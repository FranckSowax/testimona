# Lot 1 · PR 4 — Améliorer avec l'IA : plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** sur l'écran brouillon (message immédiat, annonce par e-mail), un bouton « Améliorer avec l'IA » réécrit le texte avec Claude, dans la limite de 20 appels par dossier et par jour, sans jamais envoyer les contacts ni les téléphones.

**Architecture:** une migration ajoute `ai_usage` (sans accès direct) et `consume_ai_quota` (security definer, contributeur et plus, plafond borné côté base). Un module `apps/web/src/modules/ai/` assemble des faits filtrés par liste blanche, construit le prompt (fonctions pures), consomme le quota puis appelle Claude par `@anthropic-ai/sdk` côté serveur. L'éditeur affiche le résultat sans l'enregistrer ; l'utilisateur relit puis enregistre.

**Tech Stack:** Next.js 15 (server actions), React 19, next-intl 3, Supabase (RLS, PGlite pour les tests SQL), Zod 3, Vitest, `@anthropic-ai/sdk` 1.3.

Spec : `docs/superpowers/specs/2026-10-27-lot-1-entrer-et-annoncer-design.md` (§ 1 `ai_usage`, § 3 `modules/ai`, § 4 erreurs et tests).

## Décisions de ce plan (à valider dans la PR)

- **Modèle :** `claude-opus-5-5` (modèle par défaut actuel), `output_config.effort: 'low'` (réécriture courte), réflexion adaptative laissée par défaut (elle ne peut pas être désactivée sur ce modèle). Repli serveur en cas de refus : `fallbacks: 'default'` avec la bêta `server-side-fallback-2026-07-01`. Requête non streamée, `max_tokens` 16 000, délai 45 s, une seule nouvelle tentative.
- **Plafond borné en base :** `consume_ai_quota` garde la signature de la spec (`p_daily_limit int default 20`) mais **plafonne la limite à 20** : la fonction est appelable en RPC par tout utilisateur connecté, une limite fournie par le client ne doit pas pouvoir lever le quota.
- **Jour du quota :** jour UTC.
- **Quota consommé avant l'appel**, même si l'appel échoue ensuite (règle simple, impossible à contourner en provoquant des erreurs).
- **Données envoyées (liste blanche explicite) :** prénom, nom, surnom, dates de naissance et de décès, et seulement les réponses `death.date`, `death.mentionCause`, `announce.tone`, `announce.aboutPerson`, `announce.practicalInfo`, `announce.flowersOrDonations`. Jamais `broadcast.contacts`, jamais un téléphone.
- **Lecteurs :** pas de bouton (comme « Enregistrer ») ; la base refuse de toute façon (`42501`).

`supabase/migrations/20261027000500_ai_usage.sql` :

```sql
-- ============================================================================
-- Lot 1 · PR 4 — compteur d'appels à l'IA, par dossier et par jour (UTC).
--
-- Aucun accès direct pour anon ni authenticated : seule consume_ai_quota()
-- (security definer) lit et écrit la table.
-- ============================================================================

create table public.ai_usage (
  dossier_id uuid not null references public.dossier (id) on delete cascade,
  day date not null,
  count integer not null default 0 check (count >= 0),
  primary key (dossier_id, day)
);

alter table public.ai_usage enable row level security;
-- WHY: pas de policy, et pas de privilège de table : la RLS seule laisserait
-- encore un GRANT par défaut ; les deux ensemble ferment l'accès par l'API.
revoke all on table public.ai_usage from anon, authenticated;

-- ---------------------------------------------------------------------------
-- Consomme un appel du quota du jour. Renvoie false quand le plafond est atteint.
-- WHY: security definer pour écrire dans ai_usage sans privilège de table ;
-- search_path vide contre le détournement de schéma. La limite reçue est bornée
-- à 20 : la fonction est appelable en RPC par tout utilisateur connecté, une
-- valeur fournie par le client ne doit pas pouvoir lever le quota.
-- ---------------------------------------------------------------------------
create function public.consume_ai_quota(
  p_dossier_id uuid,
  p_daily_limit integer default 20
)
returns boolean
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_limit integer := least(greatest(coalesce(p_daily_limit, 0), 0), 20);
  v_day date := (now() at time zone 'utc')::date;
  v_count integer;
begin
  if auth.uid() is null then
    raise exception 'not authenticated' using errcode = '28000';
  end if;
  if not public.is_dossier_member(p_dossier_id, 'contributor') then
    raise exception 'not allowed' using errcode = '42501';
  end if;
  if v_limit = 0 then
    return false;
  end if;

  insert into public.ai_usage as u (dossier_id, day, count)
  values (p_dossier_id, v_day, 1)
  on conflict (dossier_id, day) do update
    set count = u.count + 1
    where u.count < v_limit
  returning u.count into v_count;

  -- WHY: quand le plafond est atteint, la clause where de l'upsert n'écrit
  -- aucune ligne et v_count reste null.
  return v_count is not null;
end;
$$;

revoke execute on function public.consume_ai_quota(uuid, integer) from public, anon;
grant execute on function public.consume_ai_quota(uuid, integer) to authenticated;
```

```ts
import type { Answers } from '@/journey/progress';

/**
 * Seules réponses du parcours transmises à l'IA.
 * WHY: liste blanche explicite plutôt que « les réponses de l'étape » : une liste de diffusion
 * (contacts, téléphones) ne doit jamais partir, même si une étape future en ajoute une.
 */
export const AI_ANSWER_KEYS = [
  'death.date',
  'death.mentionCause',
  'announce.tone',
  'announce.aboutPerson',
  'announce.practicalInfo',
  'announce.flowersOrDonations',
] as const;

export type AiAnswerKey = (typeof AI_ANSWER_KEYS)[number];

/** Noms et dates de la personne ; rien d'autre de sa fiche n'est transmis. */
export interface AiPerson {
  firstName: string;
  lastName: string;
  nickname: string | null;
  birthDate: string | null;
  deathDate: string | null;
}

export interface AiFacts {
  person: AiPerson;
  answers: Partial<Record<AiAnswerKey, string>>;
}

/**
 * Faits transmis à l'IA : la personne et les réponses de la liste blanche, en texte non vide.
 *
 * @example
 * const facts = aiFacts(person, answers); // jamais broadcast.contacts
 */
export function aiFacts(person: AiPerson, answers: Answers): AiFacts {
  const picked: AiFacts['answers'] = {};
  for (const key of AI_ANSWER_KEYS) {
    const value = answers[key];
    if (typeof value === 'string' && value.trim() !== '') picked[key] = value.trim();
  }
  return {
    person: {
      firstName: person.firstName,
      lastName: person.lastName,
```

```ts
import 'server-only';
import Anthropic from '@anthropic-ai/sdk';
import { AI_MODEL } from './config';
import type { ImprovePrompt } from './prompt';

export type AiProviderFailure = 'refusal' | 'truncated' | 'empty' | 'api';

/** Échec du fournisseur ; `reason` est journalisable, jamais le texte. */
export class AiProviderError extends Error {
  constructor(readonly reason: AiProviderFailure) {
    super(`ai provider: ${reason}`);
    this.name = 'AiProviderError';
  }
}

/** Réécrit un brouillon à partir du prompt ; lève `AiProviderError` en cas d'échec. */
export type ImproveProvider = (prompt: ImprovePrompt) => Promise<string>;

// WHY: une réécriture courte à effort bas répond en quelques secondes ; 45 s et une seule
// nouvelle tentative gardent l'écran réactif sans abandonner au premier incident réseau.
const TIMEOUT_MS = 45_000;
const MAX_TOKENS = 16_000;

/**
 * Fournisseur Claude (serveur seulement). En cas de refus de sécurité, le repli serveur
 * `fallbacks: 'default'` relance la requête sur le modèle recommandé par Anthropic.
 *
 * @example
 * const text = await anthropicProvider(apiKey)(buildImprovePrompt(args));
 */
export function anthropicProvider(apiKey: string): ImproveProvider {
  const client = new Anthropic({ apiKey, timeout: TIMEOUT_MS, maxRetries: 1 });
  return async ({ system, user }) => {
    let response;
    try {
      response = await client.beta.messages.create({
        model: AI_MODEL,
        max_tokens: MAX_TOKENS,
        betas: ['server-side-fallback-2026-07-01'],
        fallbacks: 'default',
        output_config: { effort: 'low' },
        system,
        messages: [{ role: 'user', content: user }],
      });
    } catch (error) {
      // WHY: on journalise le statut, jamais le prompt ni la réponse (contenu des familles).
      if (error instanceof Anthropic.APIError) {
        console.error('[ai] provider error', { status: error.status });
      } else {
        console.error('[ai] provider error', { name: error instanceof Error ? error.name : 'unknown' });
      }
      throw new AiProviderError('api');
    }
    if (response.stop_reason === 'refusal') throw new AiProviderError('refusal');
    if (response.stop_reason === 'max_tokens') throw new AiProviderError('truncated');
    const text = response.content
      .flatMap((block) => (block.type === 'text' ? [block.text] : []))
      .join('')
      .trim();
    if (!text) throw new AiProviderError('empty');
    return text;
  };
}
```

```ts
import 'server-only';
import { z } from 'zod';
import { getStep, isStepOpen } from '@/journey/registry';
import { draftTextSchema } from '@/journey/types';
import { listAnswers } from '@/modules/dossier/answers/repo';
import { getDossier, getPerson } from '@/modules/dossier/repo';
import { isAiEnabled } from './config';
import { AI_ANSWER_KEYS, aiFacts } from './facts';
import { buildImprovePrompt } from './prompt';
import { AiProviderError, anthropicProvider, type ImproveProvider } from './provider';
import { consumeAiQuota } from './repo';
import type { AiError, AiResult } from './types';

const InputSchema = z.object({
  dossierId: z.string().uuid(),
  stepId: z.string().max(50),
  locale: z.enum(['fr', 'en']),
  text: z.string(),
});

const QUOTA_ERRORS: Record<string, AiError> = {
  '42501': 'forbidden',
  '28000': 'unauthenticated',
};

/**
 * Améliore le brouillon d'une étape avec l'IA, sans l'enregistrer.
 * Refuse une étape sans brouillon ou fermée et un texte invalide ; consomme le quota du jour
 * (contributeur et plus) avant l'appel ; n'envoie que la personne et les réponses de la liste
 * blanche. Un échec de l'IA donne `failed` : le texte affiché n'est jamais perdu.
 *
 * @example
 * const res = await improveDraft({ dossierId, stepId: 'immediate-message', locale: 'fr', text });
 */
export async function improveDraft(raw: unknown, provider?: ImproveProvider): Promise<AiResult> {
  const apiKey = process.env.ANTHROPIC_API_KEY;
  if (!isAiEnabled() || !apiKey) return { ok: false, error: 'unavailable' };

  const input = InputSchema.safeParse(raw);
  if (!input.success) return { ok: false, error: 'validation' };
  const { dossierId, stepId, locale } = input.data;
  const draft = getStep(stepId)?.draft;
  if (!draft) return { ok: false, error: 'validation' };
  const text = draftTextSchema.safeParse(input.data.text);
  if (!text.success) return { ok: false, error: 'validation' };

  const dossier = await getDossier(dossierId);
  if (!dossier) return { ok: false, error: 'not_found' };
  if (!isStepOpen(dossier.entryPoint, stepId)) return { ok: false, error: 'validation' };

  const quota = await consumeAiQuota(dossierId);
  if (quota.error) {
    const mapped = QUOTA_ERRORS[quota.error.code];
    if (!mapped) console.error('[ai] consumeAiQuota failed', { code: quota.error.code });
    return { ok: false, error: mapped ?? 'failed' };
  }
  if (!quota.data) return { ok: false, error: 'quota' };

  const [person, answers] = await Promise.all([
    getPerson(dossierId),
    listAnswers(dossierId, AI_ANSWER_KEYS),
  ]);
  // WHY: sans fiche lisible, le titre du dossier tient lieu de nom, comme sur la page brouillon.
  const facts = aiFacts(
    person ?? { firstName: '', lastName: dossier.title, nickname: null, birthDate: null, deathDate: null },
    answers,
  );
  const prompt = buildImprovePrompt({ kind: draft.kind, locale, text: text.data, facts });

  try {
    const improved = await (provider ?? anthropicProvider(apiKey))(prompt);
    const checked = draftTextSchema.safeParse(improved);
    if (!checked.success) {
      console.error('[ai] improved text rejected', { stepId });
      return { ok: false, error: 'failed' };
    }
    return { ok: true, text: checked.data };
  } catch (error) {
    const reason = error instanceof AiProviderError ? error.reason : 'unknown';
    console.error('[ai] improveDraft failed', { stepId, reason });
    return { ok: false, error: 'failed' };
  }
}
```

```tsx
'use client';

import { useTransition } from 'react';
import { useTranslations } from 'next-intl';
import { improveDraftAction } from '@/modules/ai/actions';
import type { AiError } from '@/modules/ai/types';

interface Props {
  dossierId: string;
  stepId: string;
  text: string;
  disabled: boolean;
  className: string;
  onImproved: (text: string) => void;
  onError: (error: AiError) => void;
}

/**
 * Demande une version améliorée du texte affiché ; ne l'enregistre pas.
 *
 * @example
 * <ImproveDraftButton dossierId={id} stepId={stepId} text={text} onImproved={setText} … />
 */
export function ImproveDraftButton(props: Props) {
  const t = useTranslations('journey.draft');
  const [pending, startTransition] = useTransition();
  const run = () =>
    startTransition(async () => {
      const res = await improveDraftAction({
        dossierId: props.dossierId,
        stepId: props.stepId,
        text: props.text,
      });
      if (res.ok) props.onImproved(res.text);
      else props.onError(res.error);
    });
  return (
    <button
      type="button"
      onClick={run}
      disabled={props.disabled || pending}
      aria-busy={pending || undefined}
      className={props.className}
    >
      {pending ? t('improving') : t('improve')}
    </button>
  );
}
```
